import { test, expect } from '@playwright/test';

test.describe('UNDR-06: Navigation Consistency', () => {
  const pages = ['/', '/about', '/work', '/blog', '/contact'];

  test('desktop navigation order is consistent across pages', async ({ page }) => {
    const navOrders: string[][] = [];

    for (const url of pages) {
      await page.goto(url, { waitUntil: 'domcontentloaded' });

      // Get desktop nav items (visible on desktop)
      const navItems = await page.locator('header nav[aria-label="Main navigation"] a').allTextContents();
      navOrders.push(navItems.filter(Boolean));
    }

    // All pages should have same nav order
    const firstOrder = navOrders[0];
    for (let i = 1; i < navOrders.length; i++) {
      expect(navOrders[i]).toEqual(firstOrder);
    }
  });

  test('mobile and desktop navigation have same items in same order', async ({ page }) => {
    // Set mobile viewport
    await page.setViewportSize({ width: 375, height: 667 });
    await page.goto('/', { waitUntil: 'domcontentloaded' });

    // Open mobile menu
    const menuButton = page.locator('button[aria-label*="menu" i], button[aria-label*="navigation" i]').first();
    await menuButton.click();

    // Wait for mobile menu to appear
    await page.waitForSelector('[role="dialog"]', { state: 'visible' });

    // Get mobile nav items
    const mobileNavItems = await page.locator('[role="dialog"] nav a').allTextContents();

    // Reset to desktop viewport and get desktop items
    await page.setViewportSize({ width: 1280, height: 720 });
    await page.goto('/', { waitUntil: 'domcontentloaded' });

    const desktopNavItems = await page.locator('header nav[aria-label="Main navigation"] a').allTextContents();

    // Filter and compare
    const mobileCleaned = mobileNavItems.filter(Boolean).map(s => s.trim());
    const desktopCleaned = desktopNavItems.filter(Boolean).map(s => s.trim());

    expect(mobileCleaned).toEqual(desktopCleaned);
  });

  test('active page has aria-current in desktop nav', async ({ page }) => {
    for (const url of pages) {
      await page.goto(url, { waitUntil: 'domcontentloaded' });

      // Check that exactly one nav link has aria-current="page"
      const activeLinks = page.locator('header nav a[aria-current="page"]');
      await expect(activeLinks).toHaveCount(1);
    }
  });

  test('active page has aria-current in mobile nav', async ({ page }) => {
    await page.setViewportSize({ width: 375, height: 667 });

    for (const url of pages) {
      await page.goto(url, { waitUntil: 'domcontentloaded' });

      // Open mobile menu
      const menuButton = page.locator('button[aria-label*="menu" i], button[aria-label*="navigation" i]').first();
      await menuButton.click();
      await page.waitForSelector('[role="dialog"]', { state: 'visible' });

      // Check aria-current
      const activeLinks = page.locator('[role="dialog"] nav a[aria-current="page"]');
      await expect(activeLinks).toHaveCount(1);

      // Close menu for next iteration
      await page.keyboard.press('Escape');
    }
  });
});

import { test, expect } from '@playwright/test';

test.describe('UNDR-02: Form Input Labels', () => {
  test('contact form inputs have visible labels', async ({ page }) => {
    await page.goto('/contact', { waitUntil: 'domcontentloaded' });

    // Get all input elements
    const inputs = page.locator('input:not([type="hidden"]):not([type="submit"]), textarea');
    const count = await inputs.count();

    expect(count).toBeGreaterThan(0);

    for (let i = 0; i < count; i++) {
      const input = inputs.nth(i);
      const inputId = await input.getAttribute('id');

      // Each input should have an associated label
      if (inputId) {
        const label = page.locator(`label[for="${inputId}"]`);
        await expect(label).toBeVisible();
      }
    }
  });

  test('labels are positioned above inputs (not placeholder-only)', async ({ page }) => {
    await page.goto('/contact', { waitUntil: 'domcontentloaded' });

    // Get visible labels
    const labels = page.locator('label:visible');
    const count = await labels.count();

    // There should be labels visible (not hidden or replaced by placeholders)
    expect(count).toBeGreaterThan(0);
  });
});

test.describe('UNDR-04: Required Field Indication', () => {
  test('required fields have visual asterisk', async ({ page }) => {
    await page.goto('/contact', { waitUntil: 'domcontentloaded' });

    // Check for asterisk in required field labels
    const asterisks = page.locator('label span[aria-hidden="true"]').filter({ hasText: '*' });

    // Contact form has required fields
    const count = await asterisks.count();
    expect(count).toBeGreaterThan(0);
  });

  test('required fields have screen reader text', async ({ page }) => {
    await page.goto('/contact', { waitUntil: 'domcontentloaded' });

    // Check for sr-only required text
    const srOnlyRequired = page.locator('label .sr-only, label .srOnly').filter({ hasText: /required/i });

    const count = await srOnlyRequired.count();
    expect(count).toBeGreaterThan(0);
  });
});

test.describe('UNDR-03/UNDR-05: Error Messages and Suggestions', () => {
  test('invalid email shows error message', async ({ page }) => {
    await page.goto('/contact', { waitUntil: 'domcontentloaded' });

    // Find email input
    const emailInput = page.locator('input[type="email"]');

    // Enter invalid email
    await emailInput.fill('invalid');
    await emailInput.blur();

    // Error message should appear
    const errorMessage = page.locator('[role="alert"]');
    await expect(errorMessage).toBeVisible({ timeout: 5000 });
  });

  test('email typo shows suggestion', async ({ page }) => {
    await page.goto('/contact', { waitUntil: 'domcontentloaded' });

    // Find email input
    const emailInput = page.locator('input[type="email"]');

    // Enter email with common typo
    await emailInput.fill('test@gmial.com');
    await emailInput.blur();

    // Should show "Did you mean" suggestion
    const suggestion = page.locator('text=/Did you mean|Tarkoititko|Menade du/i');
    await expect(suggestion).toBeVisible({ timeout: 5000 });
  });
});
