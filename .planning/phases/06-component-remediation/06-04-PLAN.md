**Key changes:**
1. Add `id={`tab-${tab.key}`}` - each tab needs a unique ID for aria-labelledby reference
2. Add `aria-controls={`tabpanel-${tab.key}`}` - points to the associated panel

**Remove redundant aria-disabled:**
The current code has both `aria-disabled` and `disabled`. When using `disabled` attribute, `aria-disabled` is redundant. Remove `aria-disabled={isDisabled}`.

**Note:** The tabpanel itself is NOT rendered by Tabs.tsx - it's expected to be rendered by the parent component. The Tabs component only renders the tablist. Export the expected panel ID pattern in the component documentation or as a helper function.

<task type="auto">
  <name>Task 2: Export helper function for tabpanel props</name>
  <files>nextjs-app/shared/components/Tabs/Tabs.tsx</files>
  <action>
Add a helper function that generates the correct props for a tabpanel element.

**Add after the Tabs component (before export default):**

```tsx
/**
 * Helper to generate tabpanel props for accessibility compliance.
 * Use this when rendering content panels alongside Tabs.
 *
 * @example
 * <Tabs tabs={tabs} activeTab={active} onTabChange={setActive} />
 * {tabs.map(tab => (
 *   <div {...getTabPanelProps(tab.key, tab.key === active)}>
 *     {tab.key === active && <Content />}
 *   </div>
 * ))}
 */
export function getTabPanelProps(tabKey: string, isActive: boolean) {
  return {
    id: `tabpanel-${tabKey}`,
    role: "tabpanel" as const,
    "aria-labelledby": `tab-${tabKey}`,
    hidden: !isActive,
    tabIndex: isActive ? 0 : -1,
  };
}
```

**Why hidden instead of conditional render:**
- Using `hidden` attribute (or CSS `display: none`) keeps the tabpanel in the DOM
- This maintains the aria-controls/aria-labelledby association
- Better for accessibility than conditional rendering

**Export from index.ts:** Make sure to export `getTabPanelProps` from the Tabs index file.

**Add these test cases:**

```tsx
describe("accessibility", () => {
  const tabs = [
    { key: "tab1", label: "Tab 1" },
    { key: "tab2", label: "Tab 2" },
    { key: "tab3", label: "Tab 3" },
  ];

  it("should have unique id on each tab", () => {
    render(<Tabs tabs={tabs} />);
    const tabElements = screen.getAllByRole("tab");
    const ids = tabElements.map(tab => tab.id);
    expect(new Set(ids).size).toBe(tabs.length);
    ids.forEach(id => expect(id).toMatch(/^tab-/));
  });

  it("should have aria-controls pointing to tabpanel id", () => {
    render(<Tabs tabs={tabs} />);
    const tabElements = screen.getAllByRole("tab");
    tabElements.forEach((tab, index) => {
      expect(tab).toHaveAttribute("aria-controls", `tabpanel-${tabs[index].key}`);
    });
  });

  it("should not have redundant aria-disabled when disabled", () => {
    const tabsWithDisabled = [
      { key: "tab1", label: "Tab 1" },
      { key: "tab2", label: "Tab 2", disabled: true },
    ];
    render(<Tabs tabs={tabsWithDisabled} />);
    const disabledTab = screen.getByRole("tab", { name: "Tab 2" });
    expect(disabledTab).toBeDisabled();
    // aria-disabled should not be present when disabled attribute is used
    expect(disabledTab).not.toHaveAttribute("aria-disabled");
  });
});

describe("getTabPanelProps", () => {
  it("should return correct props for active panel", () => {
    const props = getTabPanelProps("profile", true);
    expect(props).toEqual({
      id: "tabpanel-profile",
      role: "tabpanel",
      "aria-labelledby": "tab-profile",
      hidden: false,
      tabIndex: 0,
    });
  });

  it("should return correct props for inactive panel", () => {
    const props = getTabPanelProps("settings", false);
    expect(props).toEqual({
      id: "tabpanel-settings",
      role: "tabpanel",
      "aria-labelledby": "tab-settings",
      hidden: true,
      tabIndex: -1,
    });
  });
});
```
